🔄 គំរូ Function ភ្ជាប់បែប Auto-Failover (ប្តូរ Server ស្វ័យប្រវត្តិនៅពេល Error ឬយឺត) 

const API_SERVERS = [
  'https://system-mdm.onrender.com',
  'https://system-mdm-1.onrender.com',
  'https://system-mdm-2.onrender.com'
];

// Helper ហៅ API ដោយស្វ័យប្រវត្តិតាម Server ទាំង ៣
async function fetchWithFailover(endpointPath) {
  const PARTNER_KEY = 'MDM-PARTNER-2026-EXT-KEY-X9mK3pQrV7wZ';
  
  for (let i = 0; i < API_SERVERS.length; i++) {
    const serverUrl = API_SERVERS[i];
    try {
      console.log(`Trying Server ${i + 1}: ${serverUrl}...`);
      const response = await fetch(`${serverUrl}${endpointPath}`, {
        method: 'GET',
        headers: {
          'X-Partner-API-Key': PARTNER_KEY,
          'Content-Type': 'application/json'
        },
        // Timeout 8 វិនាទី បើមិនឆ្លើយតបដូរទៅ Server បន្ទាប់
        signal: AbortSignal.timeout(8000)
      });

      if (response.ok) {
        return await response.json();
      }
    } catch (err) {
      console.warn(`Server ${serverUrl} failed (${err.message}). Switching to next server...`);
    }
  }
  throw new Error('All 3 MDM API Servers are currently unreachable.');
}

// ឧទាហរណ៍នៃការប្រើប្រាស់:
async function loadData() {
  try {
    const alerts = await fetchWithFailover('/api/contracts/payment-alerts?isImmutable=true');
    console.log('Payment Alerts Received:', alerts);
  } catch (e) {
    console.error(e.message);
  }
}

import requests

API_SERVERS = [
    "https://system-mdm.onrender.com",
    "https://system-mdm-1.onrender.com",
    "https://system-mdm-2.onrender.com"
]

def fetch_with_failover(endpoint_path):
    headers = {
        "X-Partner-API-Key": "MDM-PARTNER-2026-EXT-KEY-X9mK3pQrV7wZ",
        "Content-Type": "application/json"
    }
    for server in API_SERVERS:
        url = f"{server}{endpoint_path}"
        try:
            print(f"Connecting to: {url}...")
            response = requests.get(url, headers=headers, timeout=8)
            if response.status_code == 200:
                return response.json()
        except requests.RequestException as e:
            print(f"Warning: {server} failed. Trying next server...")
            continue
    raise Exception("All 3 MDM API Servers are currently unreachable.")

<?php
function fetchWithFailover($endpointPath) {
    $servers = [
        'https://system-mdm.onrender.com',
        'https://system-mdm-1.onrender.com',
        'https://system-mdm-2.onrender.com'
    ];
    $partnerKey = 'MDM-PARTNER-2026-EXT-KEY-X9mK3pQrV7wZ';

    foreach ($servers as $server) {
        $url = $server . $endpointPath;
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 8);
        curl_setopt($ch, CURLOPT_HTTPHEADER, [
            'X-Partner-API-Key: ' . $partnerKey,
            'Content-Type: application/json'
        ]);

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($httpCode === 200 && $response) {
            return json_decode($response, true);
        }
    }
    return null;
}
